អបអរសាទរ! អ្នកបានមកដល់ចុងបញ្ចប់នៃម៉ូឌុល NumPy ហើយ។ ក្នុងនាមជា Data Analyst ថ្ងៃនេះអ្នកត្រូវបានលោកគ្រូចាត់តាំងឱ្យជួយវិភាគពិន្ទុប្រឡងឆមាសរបស់សិស្ស ចំនួន ៥០ នាក់។

## 💻 គម្រោង៖ វិភាគពិន្ទុប្រឡង

អ្នកមិនមានទិន្នន័យពិតទេ ដូច្នេះជំហានទី ១ អ្នកត្រូវបង្កើតទិន្នន័យក្លែងក្លាយសិន។ បន្ទាប់មក អ្នកត្រូវឆ្លើយសំណួរដូចជា៖ តើពិន្ទុមធ្យមប៉ុន្មាន? មានអ្នកធ្លាក់ប៉ុន្មាននាក់?

```python
import numpy as np

# កំណត់ Seed ដើម្បីអោយកូដរបស់យើងចេញលទ្ធផលដូចគ្នារហូត
np.random.seed(42)

# ១. បង្កើតពិន្ទុចៃដន្យ (ពី 30 ដល់ 100) សម្រាប់សិស្ស 50 នាក់
scores = np.random.randint(30, 101, size=50)

print("ទិន្នន័យពិន្ទុឆៅទាំង ៥០ នាក់:")
print(scores)
print("-" * 30)

# ២. វិភាគស្ថិតិរួម (Overall Statistics)
mean_score = np.mean(scores)
max_score = np.max(scores)
min_score = np.min(scores)

print(f"ពិន្ទុមធ្យមប្រចាំថ្នាក់: {mean_score:.2f}")
print(f"ពិន្ទុខ្ពស់បំផុត: {max_score}")
print(f"ពិន្ទុទាបបំផុត: {min_score}")
print("-" * 30)

# ៣. ស្វែងរកសិស្សដែលប្រឡងធ្លាក់ (ពិន្ទុ < 50)
# យើងប្រើប្រាស់ Boolean Indexing
failed_students = scores[scores < 50]
num_failed = len(failed_students) # រាប់ចំនួនអ្នកធ្លាក់

print(f"ចំនួនសិស្សដែលប្រឡងធ្លាក់ (ក្រោម ៥០ ពិន្ទុ): {num_failed} នាក់")
print(f"បញ្ជីពិន្ទុអ្នកធ្លាក់: {failed_students}")
print("-" * 30)

# ៤. ស្វែងរកសិស្សឆ្នើម (ពិន្ទុ >= 90)
excellent_students = scores[scores >= 90]
num_excellent = len(excellent_students)

print(f"ចំនួនសិស្សឆ្នើមនិទ្ទេស A (ចាប់ពី ៩០ ពិន្ទុ): {num_excellent} នាក់")
print(f"បញ្ជីពិន្ទុអ្នកនិទ្ទេស A: {excellent_students}")
print("-" * 30)

# ៥. កាត់យកពិន្ទុរបស់សិស្ស ១០ នាក់ដំបូង ដើម្បីផ្ញើអោយអាណាព្យាបាល
# យើងប្រើប្រាស់ Slicing
first_10 = scores[:10]
print(f"ពិន្ទុរបស់សិស្សលេខរៀងទី ១ ដល់ ទី ១០: {first_10}")
```
