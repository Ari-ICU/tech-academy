## 💻 ការប្រៀបធៀបដោយផ្ទាល់ដៃ

តោះសាកល្បងសរសេរកូដដើម្បីប្រៀបធៀបល្បឿននៃការគណនាលេខ ១ លានគ្រាប់!

```python
import numpy as np
import time

# បង្កើតទិន្នន័យ 1 លាន (ពីលេខ 0 ដល់ 999,999)
size = 1000000

# ១. ប្រើប្រាស់ Python List ធម្មតា
python_list = list(range(size))
start_time = time.time()
# គុណគ្រប់លេខនឹង 2 (ត្រូវប្រើ For Loop)
result_list = [x * 2 for x in python_list]
list_time = time.time() - start_time
print(f"ពេលវេលាប្រើ Python List: {list_time:.5f} វិនាទី")

# ២. ប្រើប្រាស់ NumPy Array
numpy_array = np.arange(size)
start_time = time.time()
# គុណគ្រប់លេខនឹង 2 (មិនបាច់មាន For Loop!)
result_array = numpy_array * 2
numpy_time = time.time() - start_time
print(f"ពេលវេលាប្រើ NumPy Array: {numpy_time:.5f} វិនាទី")

# ៣. ប្រៀបធៀបភាពខុសគ្នា
print(f"NumPy លឿនជាងប្រហែល {list_time/numpy_time:.0f} ដង!")
```
