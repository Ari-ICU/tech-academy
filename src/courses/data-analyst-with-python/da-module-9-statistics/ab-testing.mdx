## 🌐 សេចក្តីផ្តើម (Overview)

រាល់ពេលដែលអ្នកចូលបើក Facebook ឫ YouTube ពេលខ្លះអ្នកឃើញប៊ូតុងថ្មី ឫរចនាប័ទ្មថ្មីដែលមិត្តភក្តិរបស់អ្នកមិនមាន។ នេះមានន័យថា អ្នកកំពុងស្ថិតនៅក្នុង "ក្រុម B" នៃការធ្វើតេស្ត **A/B Testing** របស់ពួកគេហើយ! 

A/B Testing គឺជាការយក Hypothesis Testing មកអនុវត្តក្នុងពិភពជំនួញ។

## 💻 ការអនុវត្តន៍ (Code Breakdown)

ឧបមាថាក្រុមហ៊ុន E-Commerce មួយចង់ដឹងថា តើការប្តូរពាក្យលើប៊ូតុងពី "ទិញឥឡូវនេះ" ទៅជា "បន្ថែមចូលរទេះ" ធ្វើអោយអតិថិជនចុចច្រើនជាងមុនដែរឬទេ?

ពួកគេបានចែកអតិថិជនជា ២ ក្រុម (ម្នាក់ ១០០០ នាក់) ក្នុងរយៈពេល ១ សប្តាហ៍៖
- **ក្រុម A (Control):** ឃើញប៊ូតុងចាស់ "ទិញឥឡូវនេះ" (ចុចទិញ ១២០ នាក់)
- **ក្រុម B (Variant):** ឃើញប៊ូតុងថ្មី "បន្ថែមចូលរទេះ" (ចុចទិញ ១៥០ នាក់)

មើលទៅក្រុម B ដូចជាឈ្នះ! ប៉ុន្តែតើវាឈ្នះដោយសារតែ "សំណាង" ឫជា "ការពិត" (Statistically Significant)?

### ការប្រើប្រាស់ Chi-Square Test (សម្រាប់ទិន្នន័យជាភាគរយ ឫ Proportions)

ដោយសារទិន្នន័យនេះជាប្រភេទ "អ្នកចុចទិញ" និង "អ្នកអត់ចុចទិញ" (មិនមែនជាលេខប្រាក់ខែទេ) ដូច្នេះយើងត្រូវប្រើ `proportions_ztest` ឫ `chi2_contingency` ។

```python
import numpy as np
from statsmodels.stats.proportion import proportions_ztest

# ចំនួនអ្នកចុចទិញ (Successes) ក្នុងក្រុម A និង B
successes = np.array([120, 150])

# ចំនួនអ្នកមើលឃើញប៊ូតុងសរុប (Trials) ក្នុងក្រុម A និង B
trials = np.array([1000, 1000])

# ធ្វើការតេស្ត
stat, p_value = proportions_ztest(successes, trials)

print(f"P-Value គឺ: {p_value:.4f}")

# កំណត់ Alpha (កម្រិតដែលយើងទទួលយកកំហុសបានគឺ 5%)
alpha = 0.05

if p_value < alpha:
    print("✅ ប៊ូតុងថ្មីពិតជាមានប្រសិទ្ធភាពជាង (Significant)! យើងគួរប្តូរទៅប្រើប៊ូតុងថ្មី។")
else:
    print("❌ លទ្ធផលមិនច្បាស់លាស់ (Not Significant)។ ការកើនឡើងនេះអាចដោយសារចៃដន្យ គួររក្សាទុកប៊ូតុងចាស់សិន។")
```

**លទ្ធផល:**
P-Value នឹងចេញមកប្រហែល `0.045`។ ដោយសារវា **តូចជាង 0.05** យើងអាចសន្និដ្ឋានយ៉ាងមានទំនុកចិត្តថា ប៊ូតុង "បន្ថែមចូលរទេះ" ពិតជាមានឥទ្ធិពលផ្លូវចិត្តធ្វើអោយអតិថិជនចុចទិញច្រើនជាងប្រាកដមែន!
