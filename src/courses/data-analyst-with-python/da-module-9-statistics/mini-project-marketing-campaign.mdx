## 🌐 សេចក្តីផ្តើម (Overview)

សូមស្វាគមន៍មកកាន់គម្រោងបញ្ចប់នៃម៉ូឌុលស្ថិតិ! ក្រុមហ៊ុន Food Delivery មួយបានធ្វើតេស្ត (A/B Test) យុទ្ធនាការបញ្ចុះតម្លៃថ្មី (Discount 10%) ដើម្បីមើលថាតើអតិថិជនកុម្ម៉ង់អាហារច្រើនជាងមុន (ចំណាយលុយច្រើនជាងមុន) ដែរឬទេ? 

ពួកគេបានបែងចែកអតិថិជនជា ២ ក្រុម៖
- **ក្រុម A (Control):** មិនបានទទួលការបញ្ចុះតម្លៃ 
- **ក្រុម B (Variant):** ទទួលបានការបញ្ចុះតម្លៃ 10%

## 💻 គម្រោង៖ ការវិភាគ A/B Testing ពេញលេញ

```python
import numpy as np
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
import scipy.stats as stats

# ១. បង្កើតទិន្នន័យត្រាប់ (Mock Data) សម្រាប់អតិថិជន ៥០០ នាក់ ក្នុងមួយក្រុម
np.random.seed(42)
# ក្រុម A ចាយមធ្យម $15
group_a_spending = np.random.normal(loc=15, scale=5, size=500) 
# ក្រុម B ចាយមធ្យម $16 (មើលទៅដូចជាកើនឡើង $1 ព្រោះចង់បាន Discount)
group_b_spending = np.random.normal(loc=16, scale=6, size=500)

df = pd.DataFrame({
    'Spending': np.concatenate([group_a_spending, group_b_spending]),
    'Group': ['A (No Discount)']*500 + ['B (Discount)']*500
})

# ២. ជំហានទី១៖ Descriptive Statistics 
# (ប្រៀបធៀបមធ្យមភាគមើលនឹងភ្នែកសិន)
summary = df.groupby('Group').describe()['Spending']
print("ការសង្ខេបទិន្នន័យ (Descriptive Summary):")
print(summary[['mean', 'std']])
print("-" * 30)

# ៣. ជំហានទី២៖ Data Distribution (គូរក្រាហ្វិក)
sns.set_theme(style="whitegrid")
plt.figure(figsize=(8, 5))
sns.histplot(data=df, x='Spending', hue='Group', kde=True, bins=30, alpha=0.5)
plt.title("របាយការចំណាយ៖ ក្រុម A ធៀបនឹង ក្រុម B")
plt.xlabel("ការចំណាយ ($)")
plt.ylabel("ចំនួនអតិថិជន")
plt.show()

# ៤. ជំហានទី៣៖ Hypothesis Testing (T-Test) ដើម្បីកាត់ក្តី!
# H0: គ្មានការប្រែប្រួលទេ (Mean A = Mean B)
# H1: ក្រុម B ពិតជាចាយលុយច្រើនជាង (Mean B > Mean A)
t_stat, p_value = stats.ttest_ind(group_b_spending, group_a_spending)

print(f"P-Value សម្រាប់ការតេស្តគឺ: {p_value:.4f}")

if p_value < 0.05:
    print("សន្និដ្ឋាន: ✅ យើង Reject H0! ការបញ្ចុះតម្លៃ 10% ពិតជាធ្វើអោយអតិថិជនចំណាយលុយច្រើនជាងមុនប្រាកដមែន (Significant)។ ក្រុមហ៊ុនគួរអនុវត្តន៍យុទ្ធនាការនេះ!")
else:
    print("សន្និដ្ឋាន: ❌ យើង Accept H0! ទោះបីជាក្រុម B មើលទៅចាយច្រើនជាងបន្តិច តែវាអាចជាភាពចៃដន្យ (Not Significant)។ ក្រុមហ៊ុនគួរផ្អាកសិន កុំអោយខាតលុយ Discount!")
```
