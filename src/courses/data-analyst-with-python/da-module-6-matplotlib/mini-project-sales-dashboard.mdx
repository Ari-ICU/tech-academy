## 🌐 សេចក្តីផ្តើម (Overview)

សូមស្វាគមន៍មកកាន់ការបង្កើត Dashboard! ជំនួសអោយការផ្ញើរូបភាព ៤ ផ្សេងគ្នាទៅអោយថៅកែ យើងអាចរួមបញ្ចូលក្រាហ្វិកទាំង ៤ នោះទៅក្នុង "ផ្ទាំងរូបភាពតែមួយ" (ដូចផ្ទាំង Dashboard ក្នុងរថយន្តអញ្ចឹង) ដោយប្រើប្រាស់មុខងារ **Subplots**។

## 💻 គម្រោង៖ របាយការណ៍លក់ប្រចាំឆ្នាំ

```python
import pandas as pd
import matplotlib.pyplot as plt

# ១. រៀបចំទិន្នន័យ (Pandas)
data = {
    'ខែ': ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun'],
    'ចំណូល': [10, 12, 11, 15, 18, 20],      # គិតជាពាន់ដុល្លារ
    'អតិថិជនថ្មី': [50, 60, 55, 80, 100, 120]
}
df = pd.DataFrame(data)

# ២. បង្កើតស៊ុមផ្ទាំងក្រណាត់ ដែលមាន ២ ប្រអប់ (ឆ្វេង និង ស្តាំ)
# 1 ជួរដេក, 2 ជួរឈរ, ទំហំផ្ទាំងទាំងមូលធំទូលាយ (12x5)
fig, axs = plt.subplots(1, 2, figsize=(12, 5))
fig.suptitle('របាយការណ៍លក់ប្រចាំឆមាសទី១ (H1 Sales Dashboard)', fontsize=16, fontweight='bold')

# ៣. ប្រអប់ខាងឆ្វេង: axs[0] (គូរ Line Chart មើលចំណូល)
axs[0].plot(df['ខែ'], df['ចំណូល'], marker='o', color='green', linewidth=2)
axs[0].set_title('និន្នាការចំណូល ($1000)')
axs[0].set_xlabel('ខែ')
axs[0].set_ylabel('លុយចំណូល')
axs[0].grid(True, alpha=0.3)

# ៤. ប្រអប់ខាងស្តាំ: axs[1] (គូរ Bar Chart មើលចំនួនអតិថិជន)
axs[1].bar(df['ខែ'], df['អតិថិជនថ្មី'], color='skyblue')
axs[1].set_title('កំណើនអតិថិជនថ្មី')
axs[1].set_xlabel('ខែ')
axs[1].set_ylabel('ចំនួននាក់')

# ៥. កែសម្រួលគម្លាតរវាងប្រអប់កុំអោយវាសង្កត់គ្នា
plt.tight_layout()

# ៦. រក្សាទុកជាឯកសារ និងបង្ហាញ
plt.savefig('sales_dashboard_H1.png', dpi=300)
plt.show()
```
