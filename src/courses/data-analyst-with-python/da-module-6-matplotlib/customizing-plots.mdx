## 🌐 សេចក្តីផ្តើម (Overview)

នៅក្នុងពិភពធុរកិច្ច (Business) សោភ័ណភាពគឺជារឿងសំខាន់។ ក្រាហ្វិកដែលមើលទៅរញ៉េរញ៉ៃ អក្សរតូចពេក និងគ្មានពាក្យពន្យល់ នឹងធ្វើអោយថៅកែវាយតម្លៃទាបលើស្នាដៃរបស់អ្នក។ ថ្ងៃនេះយើងនឹងតុបតែងវា!

## 💻 ការបំបែកកូដ (Code Breakdown)

យើងនឹងប្រៀបធៀបចំណូលប្រចាំខែ រវាងផលិតផលពីរ (កាហ្វេ និង តែ) នៅលើផ្ទាំងតែមួយ!

```python
import matplotlib.pyplot as plt

months = ['Jan', 'Feb', 'Mar', 'Apr', 'May']
coffee_sales = [1000, 1200, 1100, 1500, 1800]
tea_sales =    [800, 900, 850, 1100, 1300]

# ១. ពង្រីកទំហំក្រាហ្វិក (ទទឹង=10, កម្ពស់=6)
plt.figure(figsize=(10, 6))

# ២. គូរបន្ទាត់កាហ្វេ (ដាក់ឈ្មោះវាថា label='Coffee' សម្រាប់ធ្វើ Legend)
# linewidth កំណត់កម្រាស់បន្ទាត់, marker កំណត់សញ្ញានៅត្រង់ចំណុចទិន្នន័យ
plt.plot(months, coffee_sales, color='brown', linewidth=2, marker='o', label='កាហ្វេ (Coffee)')

# ៣. គូរបន្ទាត់តែ នៅលើផ្ទាំងដដែល!
plt.plot(months, tea_sales, color='green', linewidth=2, marker='s', label='តែ (Tea)')

# ៤. តុបតែងចំណងជើង និងអ័ក្ស
plt.title("ការប្រៀបធៀបចំណូល: កាហ្វេ vs តែ", fontsize=16, fontweight='bold')
plt.xlabel("ខែ", fontsize=12)
plt.ylabel("ចំណូល ($)", fontsize=12)

# ៥. បើកបន្ទាត់ជំនួយ (Grid) អោយងាយស្រួលមើលលេខ
plt.grid(True, linestyle='--', alpha=0.5)

# ៦. បង្ហាញប្រអប់ពន្យល់អត្ថន័យពណ៌ (Legend) ដោយផ្អែកលើឈ្មោះ (label) ដែលយើងបានដាក់
plt.legend(loc='upper left')

# បង្ហាញក្រាហ្វិក
plt.show()
```
