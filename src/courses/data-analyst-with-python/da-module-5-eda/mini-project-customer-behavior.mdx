## 💻 គម្រោង៖ វិភាគឥរិយាបថអតិថិជន

យើងសន្មតថាមានទិន្នន័យ ដែលបានសំអាតរួចរាល់ហើយ នៅក្នុងអថេរ `df`។

```python
import pandas as pd

data = {
    'អាយុ': [22, 35, 28, 45, 21, 38, 23, 60],
    'ភេទ': ['ស្រី', 'ប្រុស', 'ស្រី', 'ប្រុស', 'ស្រី', 'ស្រី', 'ប្រុស', 'ប្រុស'],
    'ចំណាយសរុប': [150, 500, 200, 800, 120, 600, 100, 3000], # មើលលេខ 3000!
    'ថ្ងៃទិញ': ['ច័ន្ទ', 'សៅរ៍', 'សុក្រ', 'អាទិត្យ', 'ច័ន្ទ', 'សៅរ៍', 'ពុធ', 'អាទិត្យ']
}
df = pd.DataFrame(data)

print("--- ១. Univariate Analysis (មើលទូទៅ) ---")
print("ចំនួនអតិថិជនតាមភេទ:\n", df['ភេទ'].value_counts())
print("\nស្ថិតិចំណាយសរុប:\n", df['ចំណាយសរុប'].describe())
print("-" * 30)

# ចំណាំ: Max គឺ 3000 ខណៈ 75% គឺត្រឹមតែ 650។ មាន Outlier ហើយ!

print("--- ២. ចាប់ Outlier (អតិថិជន VIP) ---")
vip_customers = df[df['ចំណាយសរុប'] > 1000]
print("អតិថិជន VIP (ចំណាយលើស $1000):\n", vip_customers)
print("-" * 30)

print("--- ៣. Bivariate Analysis (រកអាថ៌កំបាំង) ---")
# ចងក្រងមើលថា តើភេទណាចំណាយច្រើនជាង?
gender_spend = df.groupby('ភេទ')['ចំណាយសរុប'].mean()
print("ចំណាយជាមធ្យមតាមភេទ:\n", gender_spend)

# ចងក្រងមើលថា តើថ្ងៃណាលក់ដាច់ជាងគេ?
day_spend = df.groupby('ថ្ងៃទិញ')['ចំណាយសរុប'].sum().sort_values(ascending=False)
print("\nចំណូលសរុបតាមថ្ងៃនៃសប្តាហ៍:\n", day_spend)
```

### 📝 របាយការណ៍ផ្ញើជូនថៅកែ (Business Insights)

ផ្អែកលើកូដខាងលើ យើងអាចសរសេររបាយការណ៍បានថា៖

1. **Insight ទី១:** ថ្ងៃចុងសប្តាហ៍ (សៅរ៍ និង អាទិត្យ) គឺជាថ្ងៃដែលរកចំណូលបានច្រើនបំផុត ខណៈថ្ងៃកណ្តាលសប្តាហ៍ (ពុធ) ស្ងាត់ខ្លាំង។
   - *Recommendation:* គួររៀបចំប្រូម៉ូសិនពិសេស "Happy Wednesday" ដើម្បីទាញភ្ញៀវនៅថ្ងៃពុធ។
2. **Insight ទី២:** យើងមានអតិថិជនបុរសវ័យ ៦០ឆ្នាំ ម្នាក់ដែលចំណាយដល់ទៅ ៣០០០ ដុល្លារ (VIP Outlier) នៅថ្ងៃអាទិត្យ។
   - *Recommendation:* គួររៀបចំកាតសមាជិក VIP ជូនគាត់ និងផ្តោតការលក់ទំនិញ Premium ទៅកាន់អតិថិជនវ័យចំណាស់។
