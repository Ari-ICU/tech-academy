## 💻 គម្រោង៖ E-Commerce Analytics

ឧបមាថាអ្នកជា Data Analyst នៅក្រុមហ៊ុន E-Commerce មួយ។ ថៅកែបានផ្តល់ទិន្នន័យ `sales_data.csv` អោយអ្នក ហើយសួរថា៖ *"តើផលិតផលមួយណាលក់ដាច់ជាងគេ? ហើយយើងគួរពង្រីកការលក់នៅខេត្តណា?"*

```python
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# ១. នាំចូលទិន្នន័យ (យើងបង្កើតទិន្នន័យក្លែងក្លាយនៅទីនេះ)
data = {
    'OrderID': [1, 2, 3, 4, 5, 6],
    'Product': ['Laptop', 'Mouse', 'Phone', 'Laptop', 'Mouse', 'Tablet'],
    'Category': ['Electronics', 'Accessories', 'Electronics', 'Electronics', 'Accessories', 'Electronics'],
    'Price': [1000, 20, 800, 1000, 20, 400],
    'Qty': [1, 5, 2, 1, 3, 1],
    'City': ['Phnom Penh', 'Siem Reap', 'Phnom Penh', 'Battambang', 'Phnom Penh', 'Siem Reap']
}
df = pd.DataFrame(data)

# ២. ការសំអាត និងរៀបចំទិន្នន័យ (Data Preparation)
# បង្កើតជួរឈរថ្មី "ចំណូលសរុប (Revenue)" = តម្លៃគុណនឹងបរិមាណ
df['Revenue'] = df['Price'] * df['Qty']

# ៣. ឆ្លើយសំណួរទី១៖ តើផលិតផលណារកលុយបានច្រើនជាងគេ? (GroupBy)
product_revenue = df.groupby('Product')['Revenue'].sum().reset_index()
product_revenue = product_revenue.sort_values(by='Revenue', ascending=False)

print("--- ចំណូលតាមផលិតផល ---")
print(product_revenue)

# ៤. ឆ្លើយសំណួរទី២៖ តើទីក្រុងណាទិញច្រើនជាងគេ? 
city_sales = df.groupby('City')['Revenue'].sum().reset_index()

# ៥. ការគូរក្រាហ្វិក (Data Visualization)
sns.set_theme(style="whitegrid")
fig, axes = plt.subplots(1, 2, figsize=(12, 5))
fig.suptitle('E-Commerce Sales Analysis', fontsize=16, fontweight='bold')

# ក្រាហ្វិកឆ្វេង: ចំណូលតាមផលិតផល
sns.barplot(data=product_revenue, x='Product', y='Revenue', ax=axes[0], palette='viridis')
axes[0].set_title('Revenue by Product')
axes[0].set_ylabel('Revenue ($)')

# ក្រាហ្វិកស្តាំ: ចំណូលតាមទីក្រុង
# រំលេចពណ៌ភ្នំពេញ
colors = ['red' if city == 'Phnom Penh' else 'grey' for city in city_sales['City']]
sns.barplot(data=city_sales, x='City', y='Revenue', ax=axes[1], palette=colors)
axes[1].set_title('Revenue by City (Phnom Penh Leads)')
axes[1].set_ylabel('Revenue ($)')

plt.tight_layout()
plt.show()
```
