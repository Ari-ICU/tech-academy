## 🌐 សេចក្តីផ្តើម (Overview)

ជំនាញ Data Analyst មិនមែនប្រើសម្រាប់តែវិភាគការលក់នោះទេ។ នាយកដ្ឋានធនធានមនុស្ស (HR) ក៏ត្រូវការអ្នកជួយខ្លាំងណាស់ដែរ! គម្រោងនេះនឹងបង្រៀនអ្នកពីវិធីជួយក្រុមហ៊ុន រក្សាបុគ្គលិកល្អៗមិនអោយលាឈប់។

## 💻 គម្រោង៖ HR Turnover Analysis

ប្រធាន HR ផ្ញើទិន្នន័យអោយអ្នក ហើយសួរថា៖ *"ឆ្នាំនេះមានបុគ្គលិកឈប់ច្រើនណាស់។ តើកត្តាអ្វីខ្លះ (ប្រាក់ខែ, ការងារច្រើនពេក, ឫអត់បានឡើងតំណែង) ដែលធ្វើអោយពួកគេលាឈប់?"*

```python
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# ១. ទិន្នន័យបុគ្គលិក
# Status: 1 = លាឈប់ (Left), 0 = នៅធ្វើការ (Stayed)
data = {
    'EmployeeID': range(1, 11),
    'Satisfaction_Score': [0.2, 0.8, 0.9, 0.3, 0.7, 0.1, 0.85, 0.4, 0.9, 0.25], # ពិន្ទុពេញ 1.0
    'Projects_Done': [6, 3, 4, 6, 3, 7, 4, 2, 3, 6],
    'Years_at_Company': [4, 2, 3, 5, 2, 4, 3, 1, 2, 5],
    'Salary_Level': ['Low', 'High', 'Medium', 'Low', 'Medium', 'Low', 'High', 'Low', 'Medium', 'Low'],
    'Status': [1, 0, 0, 1, 0, 1, 0, 1, 0, 1] # អ្នកលាឈប់គឺ 1
}
df = pd.DataFrame(data)

# ២. ស្វែងរកអត្រាលាឈប់សរុប (Turnover Rate)
turnover_rate = df['Status'].mean() * 100
print(f"អត្រាលាឈប់សរុប: {turnover_rate}%") # 50% (គ្រោះថ្នាក់ណាស់!)

# ៣. វិភាគ Correlation (តើកត្តាអ្វីទាក់ទងនឹងការលាឈប់ជាងគេ?)
# យើងយកតែទិន្នន័យលេខមកគណនា
corr_matrix = df[['Status', 'Satisfaction_Score', 'Projects_Done', 'Years_at_Company']].corr()

plt.figure(figsize=(6, 4))
sns.heatmap(corr_matrix[['Status']].sort_values(by='Status', ascending=False), 
            annot=True, cmap='coolwarm', vmin=-1, vmax=1)
plt.title("កត្តាដែលមានឥទ្ធិពលលើការលាឈប់ (Status)")
plt.show()

# ៤. វិភាគស៊ីជម្រៅលើ "ចំនួនគម្រោង (Projects Done)"
plt.figure(figsize=(7, 5))
sns.boxplot(data=df, x='Status', y='Projects_Done', palette='Set2')
plt.title("ការប្រៀបធៀបបន្ទុកការងារ (0 = នៅ, 1 = ឈប់)")
plt.xticks([0, 1], ['នៅធ្វើការ', 'លាឈប់'])
plt.ylabel('ចំនួនគម្រោងដែលត្រូវកាន់')
plt.show()
```

### 📝 របាយការណ៍អាជីវកម្ម (HR Report)

ផ្អែកលើការវិភាគ៖
1. **កម្រិតនៃភាពតានតឹង (Burnout):** Heatmap បង្ហាញថា ចំនួនគម្រោង (`Projects_Done`) មាន Correlation វិជ្ជមានខ្ពស់ (0.8+) ជាមួយនឹងការលាឈប់ (`Status`)។ ចំណែកឯភាពពេញចិត្ត (`Satisfaction_Score`) មានការថយចុះយ៉ាងខ្លាំង (Negative Correlation)។
2. **បន្ទុកការងារហួសហេតុ:** តាមរយៈ Boxplot, បុគ្គលិកដែល "លាឈប់" គឺជាអ្នកដែលត្រូវកាន់គម្រោងរហូតដល់ទៅ ៦-៧ ក្នុងពេលតែមួយ។ ចំណែកអ្នកដែលនៅធ្វើការ គឺកាន់តែ ៣-៤ ប៉ុណ្ណោះ។
3. **អនុសាសន៍ (Recommendation):** ក្រុមហ៊ុនកំពុងបាត់បង់បុគ្គលិកដែលពូកែធ្វើការ ដោយសារតែដាក់ការងារអោយពួកគេច្រើនពេក។ ប្រធាន HR គួរតែពិចារណាបែងចែកគម្រោងឡើងវិញ ឫជ្រើសរើសបុគ្គលិកថ្មីមកជួយរំលែកបន្ទុកការងារ (Load balancing) ជាបន្ទាន់។
