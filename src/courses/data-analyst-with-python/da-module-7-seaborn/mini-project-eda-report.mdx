## 💻 គម្រោង៖ របាយការណ៍ធនធានមនុស្ស

យើងនឹងចាប់ផ្តើមពីដើមដល់ចប់!

```python
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# ១. កំណត់ Theme អោយស្អាតជាមុនសិន
sns.set_theme(style="whitegrid", palette="muted")

# ២. រៀបចំទិន្នន័យ (ជាទូទៅអានពី CSV)
data = {
    'នាយកដ្ឋាន': ['IT', 'HR', 'IT', 'Marketing', 'IT', 'Marketing', 'HR', 'IT'],
    'បទពិសោធន៍(ឆ្នាំ)': [2, 5, 8, 3, 10, 1, 7, 12],
    'ប្រាក់ខែ': [600, 800, 1500, 700, 1800, 450, 1100, 2200],
    'វាយតម្លៃ': ['Good', 'Excellent', 'Excellent', 'Average', 'Excellent', 'Average', 'Good', 'Excellent']
}
df = pd.DataFrame(data)

# ៣. ឆ្លើយសំណួរទី១៖ តើនាយកដ្ឋានមួយណាដែលទទួលបានប្រាក់ខែច្រើនជាងគេ?
plt.figure(figsize=(8, 5))
# ប្រើ Barplot វាគណនាមធ្យមភាគអោយស្រេច
sns.barplot(data=df, x='នាយកដ្ឋាន', y='ប្រាក់ខែ', hue='នាយកដ្ឋាន', dodge=False)
plt.title("ប្រាក់ខែមធ្យម តាមនាយកដ្ឋាន")
plt.show()

# ៤. ឆ្លើយសំណួរទី២៖ តើបទពិសោធន៍ពិតជាមានឥទ្ធិពលលើប្រាក់ខែមែនទេ?
plt.figure(figsize=(8, 5))
# ប្រើ Scatter Plot ដើម្បីមើល Correlation
# hue='វាយតម្លៃ' ជួយបង្ហាញថា អ្នកប្រាក់ខែខ្ពស់ គឺសុទ្ធតែមានការវាយតម្លៃ Excellent ឫអត់?
sns.scatterplot(data=df, x='បទពិសោធន៍(ឆ្នាំ)', y='ប្រាក់ខែ', hue='វាយតម្លៃ', s=100)
plt.title("ទំនាក់ទំនងរវាង បទពិសោធន៍ និង ប្រាក់ខែ")
plt.show()

# ៥. ឆ្លើយសំណួរទី៣៖ តើ Correlation Matrix បង្ហាញអ្វីខ្លះ?
plt.figure(figsize=(6, 4))
# Pandas គណនា -> Seaborn គូរ
corr = df[['បទពិសោធន៍(ឆ្នាំ)', 'ប្រាក់ខែ']].corr()
sns.heatmap(corr, annot=True, cmap='coolwarm')
plt.title("Correlation Matrix")
plt.show()
```

### 📝 ការសន្និដ្ឋាន (Business Insights) ជូនប្រធាន HR

បន្ទាប់ពីមើលក្រាហ្វិកទាំង ៣ នេះរួច អ្នកអាចសរសេររបាយការណ៍ថា៖
1. នាយកដ្ឋាន IT មានកម្រិតប្រាក់ខែមធ្យមខ្ពស់ជាងគេបង្អស់។
2. បទពិសោធន៍ការងារ និងប្រាក់ខែ មានទំនាក់ទំនងស្របគ្នាស្ទើរតែ ១០០% (r = 0.98)។ មានន័យថាបទពិសោធន៍កាន់តែយូរ ប្រាក់ខែកាន់តែខ្ពស់។
3. ចំណុចសំខាន់ (Insight): ផ្អែកលើ Scatter Plot បុគ្គលិកដែលមានប្រាក់ខែខ្ពស់បំផុត មិនត្រឹមតែមានបទពិសោធន៍យូរទេ តែពួកគេសុទ្ធតែទទួលបានការវាយតម្លៃ "Excellent"។
