## 🌐 សេចក្តីផ្តើម (Overview)

សូមស្វាគមន៍មកកាន់សមរភូមិពិត! អ្នកទើបតែទទួលបានទិន្នន័យអតិថិជនពីប្រព័ន្ធចាស់របស់ក្រុមហ៊ុន។ វាមានបញ្ហាគ្រប់សព្វបែបយ៉ាង! ភារកិច្ចរបស់អ្នក គឺប្រើប្រាស់ Pandas ដើម្បីប្រែក្លាយវាទៅជាទិន្នន័យ "ស្អាត" ត្រៀមរួចជាស្រេចសម្រាប់ការវិភាគ។

## 💻 គម្រោង៖ សំអាត Database អតិថិជន

នេះគឺជាទិន្នន័យកខ្វក់ ដែលមានបញ្ហា៖ ស្ទួន (សុខ ២ដង), បាត់អាយុ, ប្រាក់ខែជាប់សញ្ញា $, និងឈ្មោះទីក្រុងវាយអក្សរធំតូចរញ៉េរញ៉ៃ។

```python
import pandas as pd
import numpy as np

# បង្កើតទិន្នន័យកខ្វក់ (Dirty Data)
data = {
    'អតិថិជន': ['សុខ', 'សៅ', 'វ៉ាន់', 'លីណា', 'សុខ'], # សុខ ស្ទួន!
    'ទីក្រុង': ['Phnom Penh', ' siem reap', 'PHNOM PENH', 'Battambang', 'Phnom Penh'], # Space ល្មើស និងអក្សរធំតូច
    'អាយុ': ['25', '30', np.nan, '22', '25'], # វ៉ាន់ បាត់អាយុ (NaN) ហើយលេខជា String
    'ប្រាក់ខែ': ['$500', '$800', '$450', '$700', '$500'] # ជាប់សញ្ញា $
}
df = pd.DataFrame(data)

print("--- មុនពេលសំអាត ---")
print(df)
print("-" * 30)

# ជំហានទី ១៖ លុបទិន្នន័យស្ទួន (Drop Duplicates)
df = df.drop_duplicates()

# ជំហានទី ២៖ សំអាតអក្សរ (String Cleaning) ក្នុងជួរឈរ "ទីក្រុង"
# កាត់ Space និងប្តូរជាអក្សរតូច
df['ទីក្រុង'] = df['ទីក្រុង'].str.strip().str.lower()
# ដើម្បីអោយស្អាត អាចប្តូរអក្សរដើមអោយធំ (Capitalize) បានទៀត
df['ទីក្រុង'] = df['ទីក្រុង'].str.title() 

# ជំហានទី ៣៖ ដោះស្រាយទិន្នន័យបាត់បង់ (Missing Values) និងបំប្លែង Type របស់ "អាយុ"
# ដោយសារ "អាយុ" ជាប់ជា String យើងត្រូវបំប្លែងជា float សិន ទើបអាចរក Mean បាន
df['អាយុ'] = df['អាយុ'].astype(float)
mean_age = df['អាយុ'].mean()
df['អាយុ'] = df['អាយុ'].fillna(mean_age).astype(int) # ចាក់បំពេញ រួចប្តូរជាលេខគត់

# ជំហានទី ៤៖ សំអាតសញ្ញា $ និងបំប្លែង Type របស់ "ប្រាក់ខែ"
df['ប្រាក់ខែ'] = df['ប្រាក់ខែ'].str.replace('$', '').astype(float)

print("--- ក្រោយពេលសំអាតរួចរាល់ ---")
print(df)
```

**លទ្ធផលរំពឹងទុក:**
```text

0     សុខ  Phnom Penh    25     500.0
1     សៅ   Siem Reap    30     800.0
2    វ៉ាន់  Phnom Penh    25     450.0  <-- អាយុត្រូវបានចាក់បំពេញ (Mean = 25.6 -> int = 25)
3    លីណា  Battambang    22     700.0
```
