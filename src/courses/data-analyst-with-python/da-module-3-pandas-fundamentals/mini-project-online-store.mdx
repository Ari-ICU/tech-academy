## 💻 គម្រោង៖ វិភាគការលក់អនឡាញ

ឧបមាថាយើងមានទិន្នន័យ `sales_data.csv` ដែលមានជួរឈរដូចជា៖ `ID`, `Customer`, `City`, `Product`, `Category`, `Price`, `Status` (ជោគជ័យ ឫ បោះបង់)។

```python
import pandas as pd

# ១. អានឯកសារ CSV ចូលមកក្នុង Pandas
# (អ្នកសន្មតថាឯកសារនេះមាននៅក្នុងកុំព្យូទ័ររួចហើយ)
# df = pd.read_csv("sales_data.csv")

# សម្រាប់ការរៀន យើងបង្កើត DataFrame គំរូមួយតូច:
data = {
    'Customer': ['សុខ', 'សៅ', 'វ៉ាន់', 'លីណា', 'រដ្ឋា'],
    'City': ['Phnom Penh', 'Siem Reap', 'Phnom Penh', 'Battambang', 'Siem Reap'],
    'Product': ['iPhone', 'T-Shirt', 'MacBook', 'T-Shirt', 'iPad'],
    'Category': ['Electronics', 'Clothing', 'Electronics', 'Clothing', 'Electronics'],
    'Price': [1000, 20, 1500, 20, 600],
    'Status': ['Completed', 'Completed', 'Cancelled', 'Completed', 'Completed']
}
df = pd.DataFrame(data)

print("ទិន្នន័យដើម (៥ជួរ):")
print(df.head())
print("-" * 30)

# ២. ការសំអាតបឋម (Filtering)
# ថៅកែមិនចង់រាប់បញ្ចូលការកុម្ម៉ង់ដែល 'Cancelled' នោះទេ ព្រោះអត់បានលុយ!
successful_sales = df[df["Status"] == "Completed"]
print(f"មានចំនួន {len(successful_sales)} ការបញ្ជាទិញដែលបានជោគជ័យ។")
print("-" * 30)

# ៣. ស្វែងរកប្រភេទទំនិញដែលលក់ដាច់ជាងគេ (GroupBy + Sorting)
# យើងប្រើប្រាស់ successful_sales ជាគោល (ព្រោះវាបានកាត់អា Cancelled ចោលហើយ)
category_revenue = successful_sales.groupby("Category")["Price"].sum()

# រៀបលំដាប់ពីធំមកតូច
top_categories = category_revenue.sort_values(ascending=False)

print("ចំណូលតាមប្រភេទទំនិញ (Top Categories):")
print(top_categories)
print("-" * 30)

# ៤. រកមើលទីក្រុងដែលមានការចំណាយច្រើនជាងគេ
city_revenue = successful_sales.groupby("City")["Price"].sum().sort_values(ascending=False)

print("ទីក្រុងដែលចំណាយច្រើនជាងគេ (Top City):")
print(city_revenue.head(1)) # ប្រើ head(1) យកតែខេត្តលេខ ១
```
