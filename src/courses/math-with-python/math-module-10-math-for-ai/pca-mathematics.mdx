## 🌐 សេចក្តីផ្តើម (Overview)

នៅក្នុងម៉ូឌុល Linear Algebra អ្នកបានរៀនអំពី Eigenvectors។ នៅក្នុងម៉ូឌុល Statistics អ្នកបានរៀនពី Covariance។ ពេលអ្នកយកវាបញ្ចូលគ្នា អ្នកនឹងទទួលបាន **PCA**! នេះជាមេរៀនដ៏កម្រិតខ្ពស់មួយ ប៉ុន្តែជាអាវុធដ៏មុតស្រួចបំផុតរបស់ Data Scientist។

## 💻 ការបំបែកកូដ (Code Breakdown)

### ដំណើរការនៃការធ្វើ PCA (ជាមួយ Scikit-Learn)

ជាជាងការសរសេរកូដរក Covariance រួចរុករក Eigenvectors ដោយខ្លួនឯង (វាវែងណាស់) ក្នុងពិភពពិត យើងប្រើប្រាស់បណ្ណាល័យ AI `scikit-learn`!

ឧបមាថាយើងមានទិន្នន័យផ្ទះ ដែលមាន ២ លក្ខណៈ (Dimension=2) គឺ "ទំហំដី" និង "ចំនួនបន្ទប់"។ ដោយសារបន្ទប់ច្រើន ភាគច្រើនមានទំហំធំ ទិន្នន័យទាំងពីរនេះត្រួតគ្នា (Redundant)។ យើងចង់បង្រួមវាអោយសល់តែ ១ វិមាត្រ (១ ជួរឈរ) បានហើយ!

```python
import numpy as np
# នាំចូល PCA ពី Scikit-learn
from sklearn.decomposition import PCA

# ទិន្នន័យដើម (៥ ផ្ទះ, ២ លក្ខណៈ)
# លក្ខណៈទី១: ទំហំ, លក្ខណៈទី២: បន្ទប់
data = np.array([
    [100, 3],
    [150, 4],
    [120, 3],
    [200, 5],
    [80,  2]
])

print("រាងទិន្នន័យដើម:", data.shape) # (5, 2) 

# ១. បង្កើតម៉ូដែល PCA ដោយប្រាប់វាថា "ខ្ញុំចង់បង្រួមអោយសល់តែ 1 លក្ខណៈ (1 Component)"
pca = PCA(n_components=1)

# ២. ឱ្យវាអនុវត្តលើទិន្នន័យរបស់យើង
data_reduced = pca.fit_transform(data)

print("រាងទិន្នន័យថ្មី:", data_reduced.shape) # (5, 1) បានបង្រួមជោគជ័យ!
print("ទិន្នន័យដែលបង្រួមរួច:\n", data_reduced)
# វាបានរួមបញ្ចូលទំហំ និងបន្ទប់ ក្លាយជាលេខអរូបីតែមួយ ដែលតំណាងអោយលក្ខណៈទាំងពីរ!

# ៣. តើយើងរក្សាទុកព័ត៌មានបានប៉ុន្មានភាគរយ?
variance_kept = pca.explained_variance_ratio_[0] * 100
print(f"យើងរក្សាអត្ថន័យដើមរបស់ទិន្នន័យបាន: {variance_kept:.2f}%")
```
